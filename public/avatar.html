btn.addEventListener('click',function(){ if(!imageData){alert('Сначала загрузите фото');return;} // Собираем полный payload var payload={service:'avatar',orientation:orientation,image:imageData}; if(currentMode==='text'){ var t=document.getElementById('txt').value.trim(); if(!t){alert('Введите текст для озвучки');return;} payload.prompt=t; payload.language=document.getElementById('lang').value; payload.voice=document.getElementById('voice').value; }else{ if(!audioData){alert('Сначала загрузите аудио');return;} payload.audio=audioData; } // Сохраняем payload в localStorage var payloadKey='seedgen_payload_'+Date.now(); localStorage.setItem(payloadKey,JSON.stringify(payload)); btn.disabled=true; statusEl.textContent='Создаём платёж...'; // Отправляем только service в pay.js (для цены) fetch('/api/pay',{ method:'POST', headers:{'Content-Type':'application/json'}, body:JSON.stringify({service:'avatar'}) }) .then(function(r){return r.json();}) .then(function(data){ if(data.error){throw new Error(data.error);} // Сохраняем связь paymentId → payloadKey localStorage.setItem('seedgen_payment_'+data.paymentId,payloadKey); window.location.href=data.confirmationUrl; }) .catch(function(e){ statusEl.textContent='Ошибка: '+e.message; btn.disabled=false; }); });
